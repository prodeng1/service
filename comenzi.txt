// Crearea unei masini (Valida, status devine DISPONIBIL)
curl -X POST http://localhost:8080/masini \
-H "Content-Type: application/json" \
-d '{"marca":"Dacia", "model":"Duster", "an":2020, "pret":15000.0, "kilometri":50000, "combustibil":"Diesel", "putereCp":115, "ownerEmail":"ion@test.ro"}'

// Crearea unei masini (Pret Foarte Mic pt Dacia - util pt a trage media in jos sau similar)
curl -X POST http://localhost:8080/masini \
-H "Content-Type: application/json" \
-d '{"marca":"Dacia", "model":"Logan", "an":2018, "pret":2000.0, "kilometri":150000, "combustibil":"Benzina", "putereCp":90, "ownerEmail":"vasile@test.ro"}'

// Creare Masina cu eroare de Rulaj (an curent, km > 1000)
curl -X POST http://localhost:8080/masini \
-H "Content-Type: application/json" \
-d '{"marca":"Tesla", "model":"Model 3", "an":2024, "pret":45000.0, "kilometri":5000, "combustibil":"Electric", "putereCp":283, "ownerEmail":"elon@test.ro"}'

// Cautarea tuturor masinilor (Poti observa campul isSuspectLowPrice la Dacia cu pret 2000 daca mai exista una de 15000)
curl -X GET http://localhost:8080/masini

// Cautarea unei masini specifice dupa ID (inlocuieste cu ID-ul real de la POST)
curl -X GET http://localhost:8080/masini/69b9cf93d83e1510f5a99652

// Actualizare status in REZERVAT (pentru testul de stergere) (inlocuieste ID)
curl -X PATCH http://localhost:8080/masini/69b9cf93d83e1510f5a99652/status \
-H "Content-Type: application/json" \
-d '{"status":"REZERVAT"}'

// Incearca stergerea unei masini REZERVATA (trebuie sa dea eroare)
curl -X DELETE http://localhost:8080/masini/69b9cf93d83e1510f5a99652

// Actualizare status in VANDUT
curl -X PATCH http://localhost:8080/masini/69b9cf93d83e1510f5a99652/status \
-H "Content-Type: application/json" \
-d '{"status":"VANDUT"}'

// Incearca editarea (PUT) unei masini VANDUTE (trebuie sa dea eroare)
curl -X PUT http://localhost:8080/masini/69b9cf93d83e1510f5a99652 \
-H "Content-Type: application/json" \
-d '{"marca":"Dacia", "model":"Duster", "an":2020, "pret":14000.0, "kilometri":50000, "combustibil":"Diesel", "putereCp":115, "ownerEmail":"ion@test.ro"}'

// Stergere masina valida (status DISPONIBIL sau VANDUT)
curl -X DELETE http://localhost:8080/masini/69b9cf93d83e1510f5a99652
